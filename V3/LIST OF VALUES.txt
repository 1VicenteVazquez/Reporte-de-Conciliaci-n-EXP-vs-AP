--LOV_NUM_INFORME
SELECT DISTINCT 
     EXPENSE_REPORT_NUM AS VALOR
    ,EXPENSE_REPORT_NUM AS DESCRIPCION
FROM 
    EXM_EXPENSE_REPORTS
WHERE 
    EXPENSE_REPORT_NUM IS NOT NULL
ORDER BY 
    EXPENSE_REPORT_NUM DESC


--LOV_RAZON_SOCIAL
SELECT DISTINCT
       XEP.NAME AS VALOR,
       XEP.NAME AS DESCRIPCION
FROM   XLE_ENTITY_PROFILES XEP,
       FUN_ALL_BUSINESS_UNITS_V FABUV
WHERE  XEP.LEGAL_ENTITY_ID = TO_NUMBER(FABUV.LEGAL_ENTITY_ID)
  AND  FABUV.BU_NAME IN ('ARCAF', 'CLID', 'PEBPE', 'PYBPY', 'UYBUY')
  AND  XEP.NAME IS NOT NULL
  AND  TRUNC(SYSDATE) BETWEEN
           NVL(XEP.EFFECTIVE_FROM, DATE '1900-01-01')
       AND NVL(XEP.EFFECTIVE_TO, DATE '4712-12-31')
ORDER BY XEP.NAME


--LOV_CUENTA_CONTABLE
SELECT 
     CONCATENATED_SEGMENTS AS DESCRIPCION_CUENTA
    ,TO_CHAR(CODE_COMBINATION_ID) AS ID_CUENTA
FROM (
    SELECT 
         GCC.CODE_COMBINATION_ID
        ,GCC.SEGMENT1 || '.' || GCC.SEGMENT2 || '.' || GCC.SEGMENT3 || '.' || GCC.SEGMENT4 AS CONCATENATED_SEGMENTS
    FROM 
        GL_CODE_COMBINATIONS GCC
    WHERE 
        GCC.ENABLED_FLAG = 'Y'
        AND TRUNC(SYSDATE) BETWEEN NVL(GCC.START_DATE_ACTIVE, TRUNC(SYSDATE)) AND NVL(GCC.END_DATE_ACTIVE, TRUNC(SYSDATE))
)
ORDER BY 
    DESCRIPCION_CUENTA


--LOV_ESTADO_AP
SELECT DISTINCT 
     APPROVAL_STATUS AS VALOR
    ,APPROVAL_STATUS AS DESCRIPCION
FROM 
    AP_INVOICES_ALL
WHERE 
    APPROVAL_STATUS IS NOT NULL
ORDER BY 
    APPROVAL_STATUS


--LOV_COLABORADOR
SELECT 
    full_name AS etiqueta, 
    person_id AS valor
FROM per_person_names_f
WHERE name_type = 'GLOBAL'
  AND TRUNC(SYSDATE) BETWEEN effective_start_date AND effective_end_date
ORDER BY full_name


--LOV_ESTADO_EXPENSES
SELECT 
    meaning AS etiqueta, 
    lookup_code AS valor
FROM fnd_lookups
WHERE lookup_type = 'EXM_REPORT_STATUS'
  AND enabled_flag = 'Y'
ORDER BY meaning


--LOV_UNIDAD_NEGOCIO
SELECT DISTINCT 
    name AS etiqueta, 
    name AS valor
FROM hr_all_organization_units_vl
WHERE name IN ('ARCAF', 'CLID', 'PEBPE', 'PYBPY', 'UYBUY')
ORDER BY name
