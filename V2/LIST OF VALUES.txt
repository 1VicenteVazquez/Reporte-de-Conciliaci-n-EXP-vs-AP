--LOV_MONEDA
SELECT 
    name || ' (' || currency_code || ')' AS etiqueta, 
    currency_code AS valor
FROM fnd_currencies_vl
WHERE enabled_flag = 'Y'
ORDER BY currency_code



--LOV_ESTADO_EXPENSES
SELECT 
    meaning AS etiqueta, 
    lookup_code AS valor
FROM fnd_lookups
WHERE lookup_type = 'EXM_REPORT_STATUS'
  AND enabled_flag = 'Y'
ORDER BY meaning



--LOV_COLABORADOR
SELECT 
    full_name AS etiqueta, 
    person_id AS valor
FROM per_person_names_f
WHERE name_type = 'GLOBAL'
  AND TRUNC(SYSDATE) BETWEEN effective_start_date AND effective_end_date
ORDER BY full_name



-LOV_UNIDAD_NEGOCIO
SELECT 
    name AS etiqueta, 
    name AS valor
FROM hr_all_organization_units_vl
WHERE name IN ('ARCAF', 'CLID', 'PEBPE', 'PYBPY', 'UYBUY')
ORDER BY name
